package main

import "fmt"

func fizzbuzz(n int) string {
	if n%15 == 0 { return "FizzBuzz" }
	if n%3 == 0 { return "Fizz" }
	if n%5 == 0 { return "Buzz" }
	return fmt.Sprintf("%d", n)
}

func main() {
	if fizzbuzz(15) != "FizzBuzz" { panic("fail 15") }
	if fizzbuzz(9) != "Fizz" { panic("fail 9") }
	if fizzbuzz(10) != "Buzz" { panic("fail 10") }
	if fizzbuzz(7) != "7" { panic("fail 7") }
	fmt.Println("PASS")
}
