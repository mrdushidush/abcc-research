package main

import (
	"fmt"
	"strings"
)

func wordCount(s string) int {
	return len(strings.Fields(s))
}

func lineCount(s string) int {
	if s == "" { return 0 }
	return len(strings.Split(s, "\n"))
}

func charCount(s string) int {
	count := 0
	for _, c := range s {
		if c != ' ' && c != '\n' && c != '\t' {
			count++
		}
	}
	return count
}

func main() {
	if wordCount("hello world") != 2 { panic("fail wordCount") }
	if wordCount("") != 0 { panic("fail empty") }
	if lineCount("a\nb\nc") != 3 { panic("fail lineCount") }
	if charCount("a b c") != 3 { panic("fail charCount") }
	fmt.Println("PASS")
}
