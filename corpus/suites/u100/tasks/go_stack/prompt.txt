package main

import "fmt"

type Stack struct {
	items []int
}

func (s *Stack) Push(item int) {
	s.items = append(s.items, item)
}

func (s *Stack) Pop() (int, bool) {
	if len(s.items) == 0 { return 0, false }
	val := s.items[len(s.items)-1]
	s.items = s.items[:len(s.items)-1]
	return val, true
}

func (s *Stack) Peek() (int, bool) {
	if len(s.items) == 0 { return 0, false }
	return s.items[len(s.items)-1], true
}

func (s *Stack) Size() int { return len(s.items) }
func (s *Stack) IsEmpty() bool { return len(s.items) == 0 }

func main() {
	s := &Stack{}
	if !s.IsEmpty() { panic("fail empty") }
	s.Push(1)
	s.Push(2)
	v, ok := s.Peek()
	if !ok || v != 2 { panic("fail peek") }
	if s.Size() != 2 { panic("fail size") }
	v, ok = s.Pop()
	if !ok || v != 2 { panic("fail pop") }
	v, ok = s.Pop()
	if !ok || v != 1 { panic("fail pop2") }
	_, ok = s.Pop()
	if ok { panic("fail empty pop") }
	fmt.Println("PASS")
}
