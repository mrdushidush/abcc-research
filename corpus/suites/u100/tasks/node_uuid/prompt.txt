Step 1: Use write_file to create uuid.js with this content:
var crypto = require('crypto');

function generateId() {
  return crypto.randomBytes(16).toString('hex');
}

function shortId() {
  return crypto.randomBytes(4).toString('hex');
}

module.exports = { generateId, shortId };
