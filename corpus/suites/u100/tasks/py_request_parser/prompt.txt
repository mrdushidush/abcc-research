import json

def parse_query_string(qs):
    params = {}
    if not qs:
        return params
    for pair in qs.split('&'):
        if '=' in pair:
            key, value = pair.split('=', 1)
            params[key] = value
    return params

def parse_json_body(body_str):
    try:
        return json.loads(body_str)
    except (json.JSONDecodeError, TypeError):
        return None
